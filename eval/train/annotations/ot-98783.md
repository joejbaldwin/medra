[[conclusion]]def solve(st):
    # Check for uniqueness
    if len(set(st)) != len(st):
        return False
    # Sort the string
    sorted_chars = sorted(st)
    # Check consecutive
    for i in range(1, len(sorted_chars)):
        if ord(sorted_chars[i]) - ord(sorted_chars[i-1]) != 1:
            return False
    return True
