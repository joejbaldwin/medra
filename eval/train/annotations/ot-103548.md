class Solution:
    def sortedCount(self, N, M, Mat):
        count = 0
        for row in Mat:
            if M == 1:
                count +=1
                continue
            # Check increasing
            is_inc = True
            for i in range(M-1):
                if row[i] >= row[i+1]:
                    is_inc = False
                    break
            if is_inc:
                count +=1
                continue
            # Check decreasing
            is_dec = True
            for i in range(M-1):
                if row[i] <= row[i+1]:
                    is_dec = False
                    break
            if is_dec:
                count +=1
        return count
