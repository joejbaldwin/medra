def main():

    import sys

    n, m, k = map(int, sys.stdin.readline().split())

    holes = set(map(int, sys.stdin.readline().split()))

    current = 1

    if current in holes:

        print(current)

        return

    for _ in range(k):

        u, v = map(int, sys.stdin.readline().split())

        if current == u:

            current = v

        elif current == v:

            current = u

        if current in holes:

            break

    print(current)
