def get_mean(arr, x, y):

    # check if x and y are valid

    n = len(arr)

    if x <=1 or y <=1:

        return -1

    if x > n or y >n:

        return -1

    # compute first x elements mean

    first_part = arr[:x]

    mean_x = sum(first_part) / x

    # compute last y elements mean

    last_part = arr[-y:]

    mean_y = sum(last_part) / y

    # compute the average of the two means

    result = (mean_x + mean_y) / 2

    return result
