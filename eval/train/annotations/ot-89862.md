def check_berlanese():
    s = input().strip()
    vowels = {'a', 'e', 'i', 'o', 'u'}
    for i in range(len(s)):
        c = s[i]
        if c not in vowels and c != 'n':
            if i == len(s) - 1:
                print("NO")
                return
            next_c = s[i+1]
            if next_c not in vowels:
                print("NO")
                return
    print("YES")

check_berlanese()
